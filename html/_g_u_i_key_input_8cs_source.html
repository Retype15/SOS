56 : base(rectT, LocalizedString.EmptyString, textAlignment, string.IsNullOrWhiteSpace(style) ? "GUIKeyInput" : style, color)

80 void CreateListener()
81 {
82 ClearListener();
83 InputListener = new GUICustomComponent(new RectTransform(Vector2.Zero, this.RectTransform),
84 onUpdate: (deltaTime, component) =>
85 {
86 var pressedKeys = PlayerInput.GetKeyboardState.GetPressedKeys();
87 if (pressedKeys?.Length > 0)
88 {
89 if (pressedKeys.Contains(Keys.Escape))
90 {
91 ClearListener();
92 return;
93 }
94
95 ApplyValue(pressedKeys.First());
96 return;
97 }
98
99 if (PlayerInput.PrimaryMouseButtonClicked() &&
100 (BGUI.MouseOn == null || BGUI.MouseOn is not GUIButton || BGUI.MouseOn.IsChildOf(this)))
101 {
102 Value = MouseButton.PrimaryMouse;
103 return;
104 }
105 else if (PlayerInput.SecondaryMouseButtonClicked())
106 {
107 Value = MouseButton.SecondaryMouse;
108 return;
109 }
110 else if (PlayerInput.MidButtonClicked())
111 {
112 Value = MouseButton.MiddleMouse;
113 return;
114 }
115 else if (PlayerInput.Mouse4ButtonClicked())
116 {
117 Value = MouseButton.MouseButton4;
118 return;
119 }
120 else if (PlayerInput.Mouse5ButtonClicked())
121 {
122 Value = MouseButton.MouseButton5;
123 return;
124 }
125 else if (PlayerInput.MouseWheelUpClicked())
126 {
127 Value = MouseButton.MouseWheelUp;
128 return;
129 }
130 else if (PlayerInput.MouseWheelDownClicked())
131 {
132 Value = MouseButton.MouseWheelDown;
133 return;
134 }
135 });
136 }
