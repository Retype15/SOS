259 Func<T, string> textSelector,
260 Func<T, Color> colorSelector)

293 public static void BindHyperlinks<T>(
294 this GUITextBlock textBlock,
295 IEnumerable<T> items,
296 Action<T> onPrimaryClick,
297 Action<T>? onSecondaryClick = null)
298 {
299 var list = items.ToList();
300
301 void ApplyLinks()
302 {
303 if (textBlock.RichTextData == null) return;
304 textBlock.ClickableAreas.Clear();
305
306 int index = 0;
307 foreach (var data in textBlock.RichTextData)
308 {
309 if (data.StartIndex >= data.EndIndex || data.StartIndex < 0 || data.Color == null) continue;
310 if (index >= list.Count) break;
311
312 var target = list[index];
313 textBlock.ClickableAreas.Add(new GUITextBlock.ClickableArea()
314 {
315 Data = data,
316 OnClick = (tb, area) => { onPrimaryClick?.Invoke(target); },
317 OnSecondaryClick = onSecondaryClick != null ? ((tb, area) => { onSecondaryClick.Invoke(target); }) : null
318 });
319 index++;
320 }
321 }
322
323 ApplyLinks();
324 textBlock.RectTransform.SizeChanged += ApplyLinks;
325 }
