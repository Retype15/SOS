254 public static RichString JoinToRichString<T>(
255 this IEnumerable<T> items,
256 string separator,
257 Func<T, string> textSelector,
258 Func<T, Color> colorSelector)
259 {
260 var sb = new System.Text.StringBuilder();
261 var list = items.ToList();
262
263 for (int i = 0; i < list.Count; i++)
264 {
265 var item = list[i];
266 Color color = colorSelector(item);
267 string text = textSelector(item).Replace("‖", "");
268
269 // Formateo simple por color.
270 sb.Append(text.SetColor(color));
271
272 if (i < list.Count - 1) sb.Append(separator);
273 }
274

291 public static void BindHyperlinks<T>(
292 this GUITextBlock textBlock,
293 IEnumerable<T> items,
294 Action<T> onPrimaryClick,
295 Action<T>? onSecondaryClick = null)
296 {
297 var list = items.ToList();
298
299 void ApplyLinks()
300 {
301 if (textBlock.RichTextData == null) return;
302 textBlock.ClickableAreas.Clear();
303
304 int index = 0;
305 foreach (var data in textBlock.RichTextData)
306 {
307 if (data.StartIndex >= data.EndIndex || data.StartIndex < 0 || data.Color == null) continue;
308 if (index >= list.Count) break;
309
310 var target = list[index];
311 textBlock.ClickableAreas.Add(new GUITextBlock.ClickableArea()
312 {
313 Data = data,
314 OnClick = (tb, area) => { onPrimaryClick?.Invoke(target); },
315 OnSecondaryClick = onSecondaryClick != null ? ((tb, area) => { onSecondaryClick.Invoke(target); }) : null
316 });
317 index++;
318 }
319 }
320
321 ApplyLinks();
322 textBlock.RectTransform.SizeChanged += ApplyLinks;
323 }

370 public static RichString FormatToXMLCode(this string rawXml)
371 {
372 if (string.IsNullOrWhiteSpace(rawXml)) return RichString.Rich("");
373
374 // 1. Scape Barotrauma RichText characters (the ‖ symbol)
375 // If the raw XML had this symbol for some reason, it would break the parser.
376 string safeXml = rawXml.Replace("‖", "||");
377
378 // 2. Highlight Strings (Values between quotes)
379 safeXml = RegexValue().Replace(safeXml, match
380 => match.Value.SetColor(ColValue));
381
382 // 3. Highlight Attributes
383 safeXml = RegexAttribute().Replace(safeXml, match
384 => match.Value.SetColor(ColAttr));
385
386 // 4. Highlight Node Names (<Node)
387 safeXml = RegexNodeName().Replace(safeXml, match
388 => match.Value.SetColor(ColNode));
389
390 // 5. Highlight Punctuation (Angles and equals signs)
391 // NOTE: To avoid replacing our own color tags (‖color:Hex‖),
392 // the regex will not touch anything between ‖ symbols.
393 // A safer way is to do punctuation first, but it interferes with HTML.
394 // To simplify, we color the basic punctuation.
395 safeXml = RegexPunctuation().Replace(safeXml, match
396 => match.Value.SetColor(ColPunctuation));
397
398 // 6. Highlight Comments (They have priority and overwrite any internal color)
399 safeXml = RegexComment().Replace(safeXml, match
400 => CleanComment().Replace(match.Value, "").SetColor(ColComment));
401
402 return safeXml.Rich();
