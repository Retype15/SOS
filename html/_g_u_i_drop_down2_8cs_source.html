310 private RectTransform FindHighestParent()
311 {
312 parentHierarchy.Clear();
313 parentHierarchy = [RectTransform.Parent];
314 RectTransform parent = parentHierarchy.Last();
315 while (parent?.Parent != null)
316 {
317 parentHierarchy.Add(parent.Parent);
318 parent = parent.Parent;
319 }
320
321 for (int i = parentHierarchy.Count - 1; i > 0; i--)
322 {
323 if (parentHierarchy[i] is GUICanvas ||
324 parentHierarchy[i].GUIComponent == null ||
325 parentHierarchy[i].GUIComponent.Style == null ||
326 parentHierarchy[i].GUIComponent == Screen.Selected?.Frame)
327 {
328 parentHierarchy.RemoveAt(i);
329 }
330 else { break; }
331 }
332 return parentHierarchy.Count > 0 ? parentHierarchy.Last() : RectTransform;
333 }
