187 private void UpdateScrollbars()
188 {
189 scrollBarsNeedsRecalculation = false;
190 if (HorizontalScrollBar == null || VerticalScrollBar == null) return;
191
192 int sbSize = ScrollBarThickness;
193 Vector4 currentPadding = paddingInitialized ? originalPadding : Padding;
194
195 float availableWidth = Rect.Width - currentPadding.X - currentPadding.Z;
196 float availableHeight = Rect.Height - currentPadding.Y - currentPadding.W;
197 float actualWidth = TextSize.X * TextScale;
198 float actualHeight = TextSize.Y * TextScale;
199
200 bool needH = actualWidth > availableWidth;
201 bool needV = actualHeight > availableHeight;
202
203 if (needH && (actualHeight > (availableHeight - sbSize))) needV = true;
204 if (needV && (actualWidth > (availableWidth - sbSize))) needH = true;
205
206 Padding = new Vector4(
207 currentPadding.X,
208 currentPadding.Y,
209 currentPadding.Z + (needV ? sbSize : 0),
210 currentPadding.W + (needH ? sbSize : 0)
211 );
212
213 HorizontalScrollBar.Visible = needH;
214 HorizontalScrollBar.Enabled = needH;
215 if (needH)
216 {
217 HorizontalScrollBar.RectTransform.Resize(new Point((int)(Rect.Width - currentPadding.X - currentPadding.Z - (needV ? sbSize : 0)), sbSize));
218 HorizontalScrollBar.RectTransform.SetPosition(Anchor.BottomLeft);
219 HorizontalScrollBar.RectTransform.AbsoluteOffset = new Point((int)currentPadding.X, (int)currentPadding.W);
220 HorizontalScrollBar.BarSize = Math.Max(0.1f, (Rect.Width - Padding.X - Padding.Z) / actualWidth);
221 }
222
223 VerticalScrollBar.Visible = needV;
224 VerticalScrollBar.Enabled = needV;
225 if (needV)
226 {
227 VerticalScrollBar.RectTransform.Resize(new Point(sbSize, (int)(Rect.Height - currentPadding.Y - currentPadding.W - (needH ? sbSize : 0))));
228 VerticalScrollBar.RectTransform.SetPosition(Anchor.TopRight);
229 VerticalScrollBar.RectTransform.AbsoluteOffset = new Point((int)currentPadding.Z, (int)currentPadding.Y);
230 VerticalScrollBar.BarSize = Math.Max(0.1f, (Rect.Height - Padding.Y - Padding.W) / actualHeight);
231 }

249 public override void Update(float deltaTime)
250 {
251 if (scrollBarsNeedsRecalculation) UpdateScrollbars();
252 base.Update(deltaTime);
253
254 if (!Visible) return;
255
256 if (BGUI.IsMouseOn(this))
257 {
258 if (ContentMenu != null && PlayerInput.SecondaryMouseButtonClicked())
259 {
260 ContentMenu(this);
261 }
262
263 if (PlayerInput.IsCtrlDown() && PlayerInput.MidButtonClicked())
264 {
265 TextScale = 0.9f;
266 scrollBarsNeedsRecalculation = true;
267 OnScaleChanged?.Invoke(TextScale);
268 return;
269 }
270
271 if (PlayerInput.ScrollWheelSpeed != 0)
272 {
273 if (PlayerInput.IsCtrlDown())
274 {
275 float oldScale = TextScale;
276 TextScale = MathHelper.Clamp(TextScale + Math.Sign(PlayerInput.ScrollWheelSpeed) * ZoomStep, MinScale, MaxScale);
277
278 if (Math.Abs(oldScale - TextScale) > 0.001f)
279 {
280 scrollBarsNeedsRecalculation = true;
281 OnScaleChanged?.Invoke(TextScale);
282 }
283 }
284 else if (PlayerInput.IsShiftDown() && HorizontalScrollBar.Enabled)
285 {
286 HorizontalScrollBar.BarScroll -= Math.Sign(PlayerInput.ScrollWheelSpeed) * 0.075f;
287 }
288 else if (VerticalScrollBar.Enabled)
289 {
290 VerticalScrollBar.BarScroll -= Math.Sign(PlayerInput.ScrollWheelSpeed) * 0.1f;
291 }
292 }
293 float maxScrollX = Math.Max(0, (TextSize.X * TextScale) - (Rect.Width - Padding.X - Padding.Z));
294 float maxScrollY = Math.Max(0, (TextSize.Y * TextScale) - (Rect.Height - Padding.Y - Padding.W));
295
296 TextOffset = new Vector2(
297 (float)Math.Round(-HorizontalScrollBar.BarScroll * maxScrollX),
298 (float)Math.Round(-VerticalScrollBar.BarScroll * maxScrollY)
299 );
300 }
