762 private static void BindHyperlinks<T>(
763 GUITextBlock textBlock,
764 IEnumerable<T> items,
765 Action<T> onPrimaryClick,
766 Action<T>? onSecondaryClick = null)
767 {
768 var list = items.ToList();
769
770 void ApplyLinks()
771 {
772 if (textBlock.RichTextData == null) return;
773 textBlock.ClickableAreas.Clear();
774
775 int index = 0;
776 foreach (var data in textBlock.RichTextData)
777 {
778 if (data.StartIndex >= data.EndIndex || data.StartIndex < 0 || data.Color == null) continue;
779 if (index >= list.Count) break;
780
781 var target = list[index];
782 textBlock.ClickableAreas.Add(new GUITextBlock.ClickableArea()
783 {
784 Data = data,
785 OnClick = (tb, area) => { onPrimaryClick?.Invoke(target); },
786 OnSecondaryClick = onSecondaryClick != null ? ((tb, area) => { onSecondaryClick.Invoke(target); }) : null
787 });
788 index++;
789 }
790 }
791
792 ApplyLinks();
793 textBlock.RectTransform.SizeChanged += ApplyLinks;
794 }
