625 private static void BindHyperlinks<T>(
626 GUITextBlock textBlock,
627 IEnumerable<T> items,
628 Action<T> onPrimaryClick,
629 Action<T>? onSecondaryClick = null)
630 {
631 var list = items.ToList();
632
633 void ApplyLinks()
634 {
635 if (textBlock.RichTextData == null) return;
636 textBlock.ClickableAreas.Clear();
637
638 int index = 0;
639 foreach (var data in textBlock.RichTextData)
640 {
641 if (data.StartIndex >= data.EndIndex || data.StartIndex < 0 || data.Color == null) continue;
642 if (index >= list.Count) break;
643
644 var target = list[index];
645 textBlock.ClickableAreas.Add(new GUITextBlock.ClickableArea()
646 {
647 Data = data,
648 OnClick = (tb, area) => { onPrimaryClick?.Invoke(target); },
649 OnSecondaryClick = onSecondaryClick != null ? ((tb, area) => { onSecondaryClick.Invoke(target); }) : null
650 });
651 index++;
652 }
653 }
654
655 ApplyLinks();
656 textBlock.RectTransform.SizeChanged += ApplyLinks;
657 }
