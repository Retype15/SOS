749 private static void BindHyperlinks<T>(
750 GUITextBlock textBlock,
751 IEnumerable<T> items,
752 Action<T> onPrimaryClick,
753 Action<T>? onSecondaryClick = null)
754 {
755 var list = items.ToList();
756
757 void ApplyLinks()
758 {
759 if (textBlock.RichTextData == null) return;
760 textBlock.ClickableAreas.Clear();
761
762 int index = 0;
763 foreach (var data in textBlock.RichTextData)
764 {
765 if (data.StartIndex >= data.EndIndex || data.StartIndex < 0 || data.Color == null) continue;
766 if (index >= list.Count) break;
767
768 var target = list[index];
769 textBlock.ClickableAreas.Add(new GUITextBlock.ClickableArea()
770 {
771 Data = data,
772 OnClick = (tb, area) => { onPrimaryClick?.Invoke(target); },
773 OnSecondaryClick = onSecondaryClick != null ? ((tb, area) => { onSecondaryClick.Invoke(target); }) : null
774 });
775 index++;
776 }
777 }
778
779 ApplyLinks();
780 textBlock.RectTransform.SizeChanged += ApplyLinks;
781 }
