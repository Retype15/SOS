233 private void ApplyInteraction(Vector2 mousePos)
234 {
235 float deltaX = mousePos.X - dragStartMouse.X;
236 float deltaY = mousePos.Y - dragStartMouse.Y;
237
238 if (Math.Abs(deltaX) < 0.01f && Math.Abs(deltaY) < 0.01f) return;
239
240 bool isCorner = IsCorner(currentDragDir);
241 bool isShift = PlayerInput.IsShiftDown();
242 bool isAlt = PlayerInput.IsAltDown();
243 bool isSymmetric = ForceSymmetric || isAlt;
244
245 if (!isCorner && isShift && !IsFixed)
246 {
247 int newX = (int)(startOffset.X + deltaX);
248 int newY = (int)(startOffset.Y + deltaY);
249
250 if (ClampToParentBounds && RectTransform.Parent != null)
251 {
252 Rectangle pRect = RectTransform.ParentRect;
253 Rectangle myRect = new(Rect.X + (newX - RectTransform.AbsoluteOffset.X),
254 Rect.Y + (newY - RectTransform.AbsoluteOffset.Y),
255 Rect.Width, Rect.Height);
256
257 if (myRect.Left < pRect.Left) newX += (pRect.Left - myRect.Left);
258 if (myRect.Right > pRect.Right) newX -= (myRect.Right - pRect.Right);
259 if (myRect.Top < pRect.Top) newY += (pRect.Top - myRect.Top);
260 if (myRect.Bottom > pRect.Bottom) newY -= (myRect.Bottom - pRect.Bottom);
261 }
262
263 RectTransform.AbsoluteOffset = new Point(newX, newY);
264 return;
265 }

267 float rawDeltaX = 0, rawDeltaY = 0;
268
269 if (currentDragDir.HasFlag(ResizeDirection.Right)) rawDeltaX = deltaX;
270 else if (currentDragDir.HasFlag(ResizeDirection.Left)) rawDeltaX = -deltaX;
271
272 if (currentDragDir.HasFlag(ResizeDirection.Bottom)) rawDeltaY = deltaY;
273 else if (currentDragDir.HasFlag(ResizeDirection.Top)) rawDeltaY = -deltaY;
274
275 if (isSymmetric) { rawDeltaX *= 2; rawDeltaY *= 2; }
276
277 float targetW = startRect.Width + rawDeltaX;
278 float targetH = startRect.Height + rawDeltaY;
279
280 bool maintainRatio = isCorner && isShift;
281 if (maintainRatio)
282 {
283 if (Math.Abs(rawDeltaX) / startRect.Width > Math.Abs(rawDeltaY) / startRect.Height)
284 targetH = targetW / startAspectRatio;
285 else
286 targetW = targetH * startAspectRatio;
287 }
288
289 Point minLimit = RectTransform.MinSize;
290 Point maxLimit = RectTransform.MaxSize;
291 int limitWX = maxLimit.X <= 0 ? int.MaxValue : maxLimit.X;
292 int limitHY = maxLimit.Y <= 0 ? int.MaxValue : maxLimit.Y;
293
294 targetW = MathHelper.Clamp(targetW, minLimit.X, limitWX);
295 targetH = MathHelper.Clamp(targetH, minLimit.Y, limitHY);
296
297 if (ClampToParentBounds && RectTransform.Parent != null)
298 {
299 Rectangle pRect = RectTransform.ParentRect;
300 float maxGrowR = pRect.Right - startRect.Right;
301 float maxGrowL = startRect.Left - pRect.Left;
302 float maxGrowB = pRect.Bottom - startRect.Bottom;
303 float maxGrowT = startRect.Top - pRect.Top;
304
305 if (currentDragDir.HasFlag(ResizeDirection.Right) && (targetW - startRect.Width) > maxGrowR) targetW = startRect.Width + maxGrowR;
306 if (currentDragDir.HasFlag(ResizeDirection.Left) && (targetW - startRect.Width) > maxGrowL) targetW = startRect.Width + maxGrowL;
307 if (currentDragDir.HasFlag(ResizeDirection.Bottom) && (targetH - startRect.Height) > maxGrowB) targetH = startRect.Height + maxGrowB;
308 if (currentDragDir.HasFlag(ResizeDirection.Top) && (targetH - startRect.Height) > maxGrowT) targetH = startRect.Height + maxGrowT;
309
310 if (maintainRatio)
311 {
312 if (targetW / startRect.Width < targetH / startRect.Height) targetH = targetW / startAspectRatio;
313 else targetW = targetH * startAspectRatio;
314 }
315 }
316
317 float finalCX = targetW - startRect.Width;
318 float finalCY = targetH - startRect.Height;
319 float moveL = 0, moveT = 0;
320
321 if (isSymmetric) { moveL = -finalCX / 2f; moveT = -finalCY / 2f; }
322 else
323 {
324 if (currentDragDir.HasFlag(ResizeDirection.Left)) moveL = -finalCX;
325 if (currentDragDir.HasFlag(ResizeDirection.Top)) moveT = -finalCY;
326 }

329 RectTransform.NonScaledSize = new Point((int)Math.Round(targetW), (int)Math.Round(targetH));
330 RectTransform.AbsoluteOffset = new Point(
331 (int)Math.Round(startOffset.X + moveL + finalCX * pivot.X),
332 (int)Math.Round(startOffset.Y + moveT + finalCY * pivot.Y)

345 private static bool IsCorner(ResizeDirection dir) =>
346 (dir.HasFlag(ResizeDirection.Left) || dir.HasFlag(ResizeDirection.Right)) &&
347 (dir.HasFlag(ResizeDirection.Top) || dir.HasFlag(ResizeDirection.Bottom));

359 private ResizeDirection GetHoverDirection(Vector2 mouse, Rectangle rect)
360 {
361 Rectangle detectRect = rect;
362 detectRect.Inflate(ResizeMargin, ResizeMargin);
363 if (!detectRect.Contains(mouse.ToPoint())) return ResizeDirection.None;
364
365 ResizeDirection dir = ResizeDirection.None;
366 if (Math.Abs(mouse.X - rect.Left) <= ResizeMargin) dir |= ResizeDirection.Left;
367 else if (Math.Abs(mouse.X - rect.Right) <= ResizeMargin) dir |= ResizeDirection.Right;
368 if (Math.Abs(mouse.Y - rect.Top) <= ResizeMargin) dir |= ResizeDirection.Top;
369 else if (Math.Abs(mouse.Y - rect.Bottom) <= ResizeMargin) dir |= ResizeDirection.Bottom;
370 return dir;
371 }

381 private static Vector2 GetPivotVector(Pivot pivot) => pivot switch

383 Pivot.TopLeft => new Vector2(0, 0),
384 Pivot.TopCenter => new Vector2(0.5f, 0),
385 Pivot.TopRight => new Vector2(1, 0),
386 Pivot.CenterLeft => new Vector2(0, 0.5f),
387 Pivot.Center => new Vector2(0.5f, 0.5f),
388 Pivot.CenterRight => new Vector2(1, 0.5f),
389 Pivot.BottomLeft => new Vector2(0, 1),
390 Pivot.BottomCenter => new Vector2(0.5f, 1),
391 Pivot.BottomRight => new Vector2(1, 1),
392 _ => Vector2.Zero
