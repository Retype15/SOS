436 {

443 {
444 Stretch = true,
445 RelativeSpacing = 0.01f,
446 CanBeFocused = false

449 var createdLists = new List<GUIListBox>(colCount);
450 for (int i = 0; i < colCount; i++)

454 Spacing = 4,

461 DrawSettings(createdLists, configs);
462 var activeLists = createdLists.Where(l => l.Content.Children.Any()).ToList();

502 public static void DrawSettings(IReadOnlyList<GUIListBox> targetLists, IEnumerable<ISOSConfig> configs)
503 {
504 var configList = configs.ToList();
505 if (targetLists.Count == 0 || configList.Count == 0) return;
506 try
507 {
508 int listCount = targetLists.Count;
509 int totalConfigs = configList.Count;
510 int chunkSize = (int)Math.Ceiling((double)totalConfigs / listCount);
511 for (int listIndex = 0; listIndex < listCount; listIndex++)
512 {
513 int startIndex = listIndex * chunkSize;
514 int count = Math.Min(chunkSize, totalConfigs - startIndex);
515 if (count <= 0) break;
516 var targetList = targetLists[listIndex];
517 for (int i = 0; i < count; i++)
518 {
519 var config = configList[startIndex + i];
520 config.DrawSettings(targetList);
521 }
522 }
523 }
524 catch (Exception ex)
525 {
526 Logger.LogDebugError($"[SOS] ProfileHelper.DrawSettings(lists) failed\n{ex}", level: LogLevel.Error);
527 }
528 }

530 private static int CalculateColumnCount(float availableWidth, int configCount, float minColumnWidth = 450f)
531 {
532 if (availableWidth <= 0 || configCount <= 0) return 1;
533 int maxColumnsByWidth = (int)Math.Floor(availableWidth / minColumnWidth);
534 return Math.Max(1, Math.Min(configCount, maxColumnsByWidth));

629 public static string GetRawXMLSafe(Prefab item)
630 {
631 var configElement = item.ConfigElement();
632 if (configElement == null) return "<!-- No XML data found for this item -->";
633 try
634 {
635 string rawXml = configElement.ToString() ?? "<!-- Empty XML -->";
636 if (rawXml == "Barotrauma.ContentXElement")
637 {
638 var type = configElement.GetType();
639 var prop = type.GetProperty("Element") ?? type.GetProperty("XElement");
640 var field = type.GetField("Element") ?? type.GetField("XElement");
641 object? inner = prop?.GetValue(configElement) ?? field?.GetValue(configElement);
642 if (inner != null) rawXml = inner.ToString() ?? rawXml;
643 }
644 return rawXml;
645 }
646 catch { return "<!-- Error parsing XML data -->"; }
647 }
