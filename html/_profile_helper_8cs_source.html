422 public static void DrawSettings(GUIComponent container, IEnumerable<ISOSConfig> configs, float minColumnWidth = MinColumnWidth)
423 {
424 if (container == null || !configs.Any()) return;
425 try
426 {
427 int colCount = CalculateColumnCount(container.Rect.Width, configs.Count(), minColumnWidth);
428 _currentColumnCount = colCount;
429 var layoutGroup = new GUILayoutGroup(new RectTransform(Vector2.One, container.RectTransform), isHorizontal: true)
430 {
431 Stretch = true,
432 RelativeSpacing = 0.01f,
433 CanBeFocused = false
434 };
435 float relWidth = 1.0f / colCount;
436 var createdLists = new List<GUIListBox>(colCount);
437 for (int i = 0; i < colCount; i++)
438 {
439 var list = new GUIListBox(new RectTransform(new Vector2(relWidth, 1f), layoutGroup.RectTransform), style: null)
440 {
441 Spacing = 4,
442 Padding = new Vector4(8, 8, 8, 8),
443 CanBeFocused = true
444 };
445 createdLists.Add(list);
446 }
447 _contentLists = createdLists;
448 DrawSettings(createdLists, configs);
449 var activeLists = createdLists.Where(l => l.Content.Children.Any()).ToList();
450 if (activeLists.Count < createdLists.Count)
451 {
452 foreach (var emptyList in createdLists.Where(l => !l.Content.Children.Any()))
453 {
454 emptyList.RemoveFromGUIUpdateList();
455 layoutGroup.RemoveChild(emptyList);
456 }
457 if (activeLists.Count > 0)
458 {
459 float adjustedWidth = 1.0f / activeLists.Count;
460 foreach (var list in activeLists)
461 list.RectTransform.RelativeSize = new Vector2(adjustedWidth, 1f);
462 }
463 layoutGroup.Recalculate();
464 }
465 }
466 catch (Exception ex)
467 {
468 Logger.LogDebugError($"[SOS] ProfileHelper.DrawSettings(container) failed\n{ex}", level: LogLevel.Error);
469 }

489 public static void DrawSettings(IReadOnlyList<GUIListBox> targetLists, IEnumerable<ISOSConfig> configs)
490 {
491 var configList = configs.ToList();
492 if (targetLists.Count == 0 || configList.Count == 0) return;
493 try
494 {
495 int listCount = targetLists.Count;
496 int totalConfigs = configList.Count;
497 int chunkSize = (int)Math.Ceiling((double)totalConfigs / listCount);
498 for (int listIndex = 0; listIndex < listCount; listIndex++)
499 {
500 int startIndex = listIndex * chunkSize;
501 int count = Math.Min(chunkSize, totalConfigs - startIndex);
502 if (count <= 0) break;
503 var targetList = targetLists[listIndex];
504 for (int i = 0; i < count; i++)
505 {
506 var config = configList[startIndex + i];
507 ProfileHelper.TryDraw(targetList.Content.RectTransform, config);
508 }
509 }
510 }
511 catch (Exception ex)
512 {
513 Logger.LogDebugError($"[SOS] ProfileHelper.DrawSettings(lists) failed\n{ex}", level: LogLevel.Error);
514 }
515 }

517 private static int CalculateColumnCount(float availableWidth, int configCount, float minColumnWidth = 450f)
518 {
519 if (availableWidth <= 0 || configCount <= 0) return 1;
520 int maxColumnsByWidth = (int)Math.Floor(availableWidth / minColumnWidth);
521 return Math.Max(1, Math.Min(configCount, maxColumnsByWidth));
522 }

645 {
646 var rectT = new RectTransform(new Vector2(1f, 0f), parent, Anchor.TopCenter);
647 if (typeName.IsNullOrEmpty()) typeName = config.GetType().FullOrName();
648 try
649 {
650 config.Draw(rectT);
651 if (rectT.CountChildren == 0)
652 {
653 Logger.LogDebug($"'{typeName}' have nothing to draw, removing created RectTransform...", level: LogLevel.Trace);
654 rectT.Parent = null;
655 return false;
656 }
657 return true;

661 Logger.LogError($"[SOS] Exception when try to draw '{typeName}': {ex.Message}");
662 if (rectT.CountChildren == 0) rectT.Parent = null;

716 public static string GetRawXMLSafe(Prefab item)
717 {
718 var configElement = item.ConfigElement();
719 if (configElement == null) return "<!-- No XML data found for this item -->";
720 try
721 {
722 string rawXml = configElement.ToString() ?? "<!-- Empty XML -->";
723 if (rawXml == "Barotrauma.ContentXElement")
724 {
725 var type = configElement.GetType();
726 var prop = type.GetProperty("Element") ?? type.GetProperty("XElement");
727 var field = type.GetField("Element") ?? type.GetField("XElement");
728 object? inner = prop?.GetValue(configElement) ?? field?.GetValue(configElement);
729 if (inner != null) rawXml = inner.ToString() ?? rawXml;
730 }
731 return rawXml;
732 }
733 catch { return "<!-- Error parsing XML data -->"; }
734 }
