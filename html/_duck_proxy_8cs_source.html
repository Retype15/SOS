204 private void ConfigureClrObject(object target)
205 {
206 // Subscribe the target's dispose affordance once (public or explicit). Absent => silent no-op.
207 _disposeAction = target is IDisposable disposable ? disposable.Dispose : null;
208
209 var targetType = target.GetType();
210 var interfaceType = typeof(T);
211 var errors = new StringBuilder();
212
213 foreach (var interfaceMethod in GetAllInterfaceMethods(interfaceType))
214 {
215 if (IsDisposeMethod(interfaceMethod))
216 {
217 _handlerMap[interfaceMethod] = _ => { _disposeAction?.Invoke(); return null; };
218 continue;
219 }
220
221 var parameterTypes = interfaceMethod.GetParameters()
222 .Select(p => p.ParameterType)
223 .ToArray();
224
225 var targetMethod = targetType.GetMethod(interfaceMethod.Name, parameterTypes);
226
227 if (targetMethod == null)
228 {
229 if (TryResolveFallback(interfaceMethod, out var fallbackHandler))
230 _handlerMap[interfaceMethod] = fallbackHandler!;
231 else
232 errors.AppendLine($" Missing method or property: '{interfaceMethod.Name}'");
233
234 continue;
235 }
236
237 if (targetMethod.ReturnType != interfaceMethod.ReturnType)
238 {
239 errors.AppendLine($" Return type mismatch on '{interfaceMethod.Name}'. Expected: {interfaceMethod.ReturnType.Name}, Found: {targetMethod.ReturnType.Name}");
240 continue;
241 }
242
243 _handlerMap[interfaceMethod] = args => targetMethod.Invoke(target, args);
244 }
245
246 if (errors.Length > 0)
247 {
248 throw new InvalidCastException(
249 $"Type '{targetType.FullName}' does not satisfy contract of '{interfaceType.FullName}':\n" +
250 errors.ToString());
251 }

254 private void ConfigureLuaTable(Table table)
255 {
256 var disposeFunc = table.Get("Dispose");
257 if (disposeFunc.Type == DataType.Function || disposeFunc.Type == DataType.ClrFunction)
258 _disposeAction = () => table.OwnerScript.Call(disposeFunc);
259
260 var interfaceType = typeof(T);
261 var errors = new StringBuilder();
262
263 foreach (var interfaceMethod in GetAllInterfaceMethods(interfaceType))
264 {
265 if (IsDisposeMethod(interfaceMethod))
266 {
267 _handlerMap[interfaceMethod] = _ => { _disposeAction?.Invoke(); return null; };
268 continue;
269 }
270
271 var methodName = interfaceMethod.Name;
272 var returnType = interfaceMethod.ReturnType;
273
274 if (methodName.StartsWith("get_", ignoreCase: true, null))
275 {
276 var propName = methodName[4..];
277
278 if (table.Get(propName).IsNil())
279 {
280 if (TryResolveFallback(interfaceMethod, out var fallbackHandler))
281 _handlerMap[interfaceMethod] = fallbackHandler!;
282 else
283 errors.AppendLine($" Missing property getter '{propName}' in Lua table.");
284
285 continue;
286 }
287
288 _handlerMap[interfaceMethod] = _ => table.Get(propName).ToObject(returnType);
289 }
290 else if (methodName.StartsWith("set_", ignoreCase: true, null))
291 {
292 var propName = methodName[4..];
293
294 if (table.Get(propName).IsNil())
295 {
296 if (TryResolveFallback(interfaceMethod, out var fallbackHandler))
297 _handlerMap[interfaceMethod] = fallbackHandler!;
298 else
299 errors.AppendLine($" Missing property setter '{propName}' in Lua table.");
300
301 continue;
302 }
303
304 _handlerMap[interfaceMethod] = args =>
305 {
306 table.Set(propName, DynValue.FromObject(table.OwnerScript, args.FirstOrDefault()));
307 return null;
308 };
309 }
310 else
311 {
312 var luaFunc = table.Get(methodName);
313
314 if (luaFunc.Type != DataType.Function && luaFunc.Type != DataType.ClrFunction)
315 {
316 if (TryResolveFallback(interfaceMethod, out var fallbackHandler))
317 _handlerMap[interfaceMethod] = fallbackHandler!;
318 else
319 errors.AppendLine($" Missing function '{methodName}' in Lua table.");
320
321 continue;
322 }
323
324 _handlerMap[interfaceMethod] = args =>
325 {
326 var result = table.OwnerScript.Call(luaFunc, args);
327 return returnType == typeof(void) ? null : result.ToObject(returnType);
328 };
329 }
330 }
331
332 if (errors.Length > 0)
333 {
334 throw new InvalidCastException(
335 $"Lua table does not satisfy contract of '{interfaceType.FullName}':\n" +
336 errors.ToString());
337 }
338 }

348 private static bool TryResolveFallback(MethodInfo interfaceMethod, out Func<object?[], object?>? handler)
349 {
350 handler = null;
351 var fallbackAttr = interfaceMethod.GetCustomAttribute<DefaultClassAttributeBase>();
352
353 if (fallbackAttr == null)
354 {
355 var declaringType = interfaceMethod.DeclaringType;
356 if (declaringType != null)
357 {
358 var property = declaringType.GetProperties()
359 .FirstOrDefault(p => p.GetMethod == interfaceMethod);
360 fallbackAttr = property?.GetCustomAttribute<DefaultClassAttributeBase>();
361 }
362 }
363
364 if (fallbackAttr == null)
365 {
366 var declaringType = interfaceMethod.DeclaringType;
367 if (declaringType != null)
368 {
369 fallbackAttr = declaringType.GetCustomAttribute<DefaultClassAttributeBase>(inherit: true);
370 }
371 }
372
373 if (fallbackAttr == null) return false;
374
375 var staticFallbackMethod = fallbackAttr.GetMethod(interfaceMethod);
376 if (staticFallbackMethod == null) return false;
377
378 handler = args => staticFallbackMethod.Invoke(null, args);
379 return true;
380 }
