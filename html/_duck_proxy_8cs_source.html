213 private void ConfigureLuaTable(Table table)
214 {
215 var interfaceType = typeof(T);
216 var errors = new List<string>();
217
218 foreach (var interfaceMethod in GetAllInterfaceMethods(interfaceType))
219 {
220 var methodName = interfaceMethod.Name;
221 var returnType = interfaceMethod.ReturnType;
222
223 if (methodName.StartsWith("get_"))
224 {
225 var propName = methodName[4..];
226
227
228 if (table.Get(propName).IsNil())
229 {
230 if (TryResolveFallback(interfaceMethod, out var fallbackHandler))
231 _handlerMap[interfaceMethod] = fallbackHandler!;
232 else
233 errors.Add($"Missing property '{propName}' in Lua table.");
234
235 continue;
236 }
237
238 _handlerMap[interfaceMethod] = _ => table.Get(propName).ToObject(returnType);
239 }
240 else if (methodName.StartsWith("set_"))
241 {
242 var propName = methodName[4..];
243
244 _handlerMap[interfaceMethod] = args =>
245 {
246 table.Set(propName, DynValue.FromObject(table.OwnerScript, args.FirstOrDefault()));
247 return null;
248 };
249 }
250 else
251 {
252 var luaFunc = table.Get(methodName);
253
254 if (luaFunc.Type != DataType.Function && luaFunc.Type != DataType.ClrFunction)
255 {
256 if (TryResolveFallback(interfaceMethod, out var fallbackHandler))
257 _handlerMap[interfaceMethod] = fallbackHandler!;
258 else
259 errors.Add($"Missing function '{methodName}' in Lua table.");
260
261 continue;
262 }
263
264 _handlerMap[interfaceMethod] = args =>
265 {
266 var result = table.OwnerScript.Call(luaFunc, args);
267 return returnType == typeof(void) ? null : result.ToObject(returnType);
268 };
269 }
270 }
271
272 if (errors.Count > 0)
273 {
274 throw new InvalidCastException(
275 $"Lua table does not satisfy contract of '{interfaceType.FullName}':\n" +
276 string.Join("\n", errors));
277 }

288 private static bool TryResolveFallback(MethodInfo interfaceMethod, out Func<object?[], object?>? handler)
289 {
290 handler = null;
291 var fallbackAttr = interfaceMethod.GetCustomAttribute<DefaultClassAttributeBase>();
292
293 if (fallbackAttr == null)
294 {
295 var declaringType = interfaceMethod.DeclaringType;
296 if (declaringType != null)
297 {
298 var property = declaringType.GetProperties()
299 .FirstOrDefault(p => p.GetMethod == interfaceMethod);
300 fallbackAttr = property?.GetCustomAttribute<DefaultClassAttributeBase>();
301 }
302 }
303
304 if (fallbackAttr == null)
305 {
306 var declaringType = interfaceMethod.DeclaringType;
307 if (declaringType != null)
308 {
309 fallbackAttr = declaringType.GetCustomAttribute<DefaultClassAttributeBase>(inherit: true);
310 }
311 }
312
313 if (fallbackAttr == null) return false;
314
315 var staticFallbackMethod = fallbackAttr.GetMethod(interfaceMethod);
316 if (staticFallbackMethod == null) return false;

318 handler = args => staticFallbackMethod.Invoke(null, args);
