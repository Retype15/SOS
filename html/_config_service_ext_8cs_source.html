54 public static Result SaveBatch(this ConfigService configService, IReadOnlyCollection<ISettingBase> settings)
55 {
56 if (settings.Count == 0) return Result.Ok();
57
58 Result? resultLog = new();
59 var groups = settings.GroupBy((setting) => setting.OwnerPackage);
60 foreach (var settingsBatch in groups)
61 {
62 XDocument cpCfgValues;
63 var package = settingsBatch.Key;
64
65 if (configService._storageService.LoadLocalXml(package, SaveDataFileName) is not { } saveFileResult)
66 return Result.Fail($"{nameof(SaveBatch)}: Storage Service Failure while trying to load file for package [{package.Name}].");
67
68 if (saveFileResult.IsSuccess)
69 cpCfgValues = saveFileResult.Value;
70 else
71 cpCfgValues = new XDocument(new XDeclaration("1.0", "utf-8", "yes"), new XElement("Configuration"));
72
73 if (cpCfgValues.Root is null || cpCfgValues.Root.Name != "Configuration")
74 return Result.Fail($"{nameof(SaveBatch)}: Bad save file format for package: [{package.Name}]");
75
76 var mainTarget = GetOrAddElement(cpCfgValues.Root, XmlConvert.EncodeLocalName(package.Name.Trim()), name => new XElement(name));
77
78 foreach (var setting in settingsBatch)
79 {
80 var currentTarget = GetOrAddElement(mainTarget, setting.InternalName, name => new XElement(name));
81
82 var ret = setting.GetSerializableValue().Match<bool>(str =>
83 {
84 var tgt = currentTarget.Attribute("Value");
85 if (tgt is null)
86 {
87 var attr = new XAttribute("Value", str);
88 currentTarget.Add(attr);
89 }
90 else
91 {
92 tgt.Value = str;
93 }
94
95 return true;
96 },
97 elem =>
98 {
99 currentTarget.ReplaceNodes(new XElement("Value", elem));
100 return true;
101 });
102
103 //resultLog.WithReasons(ret.Reasons);
104 }
105
106 resultLog.WithReasons(configService._storageService.SaveLocalXml(package, SaveDataFileName, cpCfgValues).Reasons);
107 }
108
109 return resultLog;
110 }

121 static XElement GetOrAddElement(XElement containerElement, string elementName, Func<string, XElement> factory)
122 {
123 var element = containerElement.Element(elementName);
124 if (element is null)
125 {
126 element = factory(elementName);
127 containerElement.Add(element);
128 }
129 return element;
130 }
