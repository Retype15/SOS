648 t += CoroutineManager.DeltaTime;

717 private IEnumerable<CoroutineStatus> ExecuteSequence()
718 {
719 yield return CoroutineStatus.Running;
720
721 int stepIndex = 0;
722 while (stepIndex < steps.Count)
723 {
724 var step = steps[stepIndex];
725
726 var activeCoroutines = new List<IEnumerator<CoroutineStatus>>();
727 foreach (var action in step)
728 {
729 activeCoroutines.Add(action().GetEnumerator());
730 }
731
732 while (activeCoroutines.Count > 0)
733 {
734 for (int i = activeCoroutines.Count - 1; i >= 0; i--)
735 {
736 if (!activeCoroutines[i].MoveNext() || activeCoroutines[i].Current == CoroutineStatus.Success)
737 {
738 activeCoroutines.RemoveAt(i);
739 }
740 }
741
742 if (activeCoroutines.Count > 0)
743 {
744 yield return CoroutineStatus.Running;
745 }
746 }
747
748 stepIndex++;
749 }
750
751 yield return CoroutineStatus.Success;
